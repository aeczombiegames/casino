import pygame import random import time # Initialize Pygame pygame.init() # Define constants SCREEN_WIDTH, SCREEN_HEIGHT = 800, 600 REEL_WIDTH, REEL_HEIGHT = 100, 100 SYMBOLS = ['🍒', '🍋', '🍇', '🍉', '🔔', 'BAR', 'JACKPOT'] FONT = pygame.font.SysFont('Arial', 36) # Set up the screen screen = pygame.display.set_mode((SCREEN_WIDTH, SCREEN_HEIGHT)) pygame.display.set_caption("Slot Machine") # Colors WHITE = (255, 255, 255) GREEN = (0, 255, 0) BLACK = (0, 0, 0) # Load images for the symbols symbol_images = {} for symbol in SYMBOLS: symbol_images[symbol] = FONT.render(symbol, True, BLACK) # Function to draw reels def draw_reels(reels): for i, symbol in enumerate(reels): pygame.draw.rect(screen, WHITE, (200 + i * (REEL_WIDTH + 10), 200, REEL_WIDTH, REEL_HEIGHT)) screen.blit(symbol_images[symbol], (200 + i * (REEL_WIDTH + 10) + 20, 230)) # Spin the reels and return 3 random symbols def spin_reels(): return random.choices(SYMBOLS, k=3) # Function to check payouts def check_payout(reels): if reels[0] == reels[1] == reels[2]: return 100 # Win 100 coins for matching symbols return 0 # Function to animate the reels def animate_reels(): reels = [random.choice(SYMBOLS) for _ in range(3)] spin_duration = 1 # Duration of the spin in seconds frames = 30 spin_speed = spin_duration / frames for _ in range(frames): screen.fill(GREEN) # Draw spinning reels with animation for i in range(3): reels[i] = random.choice(SYMBOLS) # Randomly change symbols for animation pygame.draw.rect(screen, WHITE, (200 + i * (REEL_WIDTH + 10), 200, REEL_WIDTH, REEL_HEIGHT)) screen.blit(symbol_images[reels[i]], (200 + i * (REEL_WIDTH + 10) + 20, 230)) pygame.display.update() time.sleep(spin_speed) # Wait for the next frame return reels # Main game loop def game_loop(): running = True reels = ['🍒', '🍋', '🍇'] # Initial state of the reels balance = 1000 # Player's starting balance bet_options = [10, 20, 50, 100] bet = bet_options[0] # Default bet selected_bet = bet while running: screen.fill(GREEN) for event in pygame.event.get(): if event.type == pygame.QUIT: running = False elif event.type == pygame.MOUSEBUTTONDOWN: # Spin the reels when clicked reels = animate_reels() # Animate the reels payout = check_payout(reels) balance += payout - selected_bet # Update balance (subtract bet, add payout) # Draw the reels draw_reels(reels) # Draw the balance balance_text = FONT.render(f"Balance: ${balance}", True, BLACK) screen.blit(balance_text, (SCREEN_WIDTH // 2 - balance_text.get_width() // 2, 50)) # Draw the spin button pygame.draw.rect(screen, BLACK, (SCREEN_WIDTH // 2 - 100, 450, 200, 50)) spin_button_text = FONT.render("SPIN", True, WHITE) screen.blit(spin_button_text, (SCREEN_WIDTH // 2 - spin_button_text.get_width() // 2, 460)) # Draw the bet options bet_text = FONT.render(f"Bet: ${selected_bet}", True, BLACK) screen.blit(bet_text, (SCREEN_WIDTH // 2 - bet_text.get_width() // 2, 120)) # Draw bet selection buttons for i, option in enumerate(bet_options): bet_button_rect = pygame.Rect(50 + i * 150, 350, 100, 50) pygame.draw.rect(screen, WHITE, bet_button_rect) bet_button_text = FONT.render(f"${option}", True, BLACK) screen.blit(bet_button_text, (bet_button_rect.centerx - bet_button_text.get_width() // 2, bet_button_rect.centery - bet_button_text.get_height() // 2)) # Check for bet selection if bet_button_rect.collidepoint(pygame.mouse.get_pos()) and pygame.mouse.get_pressed()[0]: selected_bet = option pygame.display.update() # Start the game game_loop() # Quit Pygame pygame.quit()
