import pygame import random # Initialize Pygame pygame.init() # Define constants SCREEN_WIDTH, SCREEN_HEIGHT = 800, 600 REEL_WIDTH, REEL_HEIGHT = 100, 100 SYMBOLS = ['🍒', '🍋', '🍇', '🍉', '🔔', 'BAR', 'JACKPOT'] FONT = pygame.font.SysFont('Arial', 36) # Set up the screen screen = pygame.display.set_mode((SCREEN_WIDTH, SCREEN_HEIGHT)) pygame.display.set_caption("Slot Machine") # Colors WHITE = (255, 255, 255) GREEN = (0, 255, 0) BLACK = (0, 0, 0) # Load images for the symbols symbol_images = {} for symbol in SYMBOLS: symbol_images[symbol] = FONT.render(symbol, True, BLACK) # Function to draw reels def draw_reels(reels): for i, symbol in enumerate(reels): pygame.draw.rect(screen, WHITE, (200 + i * (REEL_WIDTH + 10), 200, REEL_WIDTH, REEL_HEIGHT)) screen.blit(symbol_images[symbol], (200 + i * (REEL_WIDTH + 10) + 20, 230)) # Spin the reels and return 3 random symbols def spin_reels(): return random.choices(SYMBOLS, k=3) # Function to check payouts def check_payout(reels): if reels[0] == reels[1] == reels[2]: return 100 # Win 100 coins for matching symbols return 0 # Main game loop def game_loop(): running = True reels = ['🍒', '🍋', '🍇'] # Initial state of the reels balance = 1000 # Player's starting balance bet = 50 # Starting bet while running: screen.fill(GREEN) for event in pygame.event.get(): if event.type == pygame.QUIT: running = False elif event.type == pygame.MOUSEBUTTONDOWN: # Spin the reels when clicked reels = spin_reels() payout = check_payout(reels) balance += payout - bet # Update balance (subtract bet, add payout) # Draw the reels draw_reels(reels) # Draw the balance balance_text = FONT.render(f"Balance: ${balance}", True, BLACK) screen.blit(balance_text, (SCREEN_WIDTH // 2 - balance_text.get_width() // 2, 50)) # Draw the spin button pygame.draw.rect(screen, BLACK, (SCREEN_WIDTH // 2 - 100, 450, 200, 50)) spin_button_text = FONT.render("SPIN", True, WHITE) screen.blit(spin_button_text, (SCREEN_WIDTH // 2 - spin_button_text.get_width() // 2, 460)) pygame.display.update() # Start the game game_loop() # Quit Pygame pygame.quit()
